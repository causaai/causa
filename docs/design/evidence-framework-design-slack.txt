📋 EVIDENCE FRAMEWORK DESIGN

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

🎯 PROBLEM

Current (RCA Pipeline):
  • Evidence = plain strings: ["Exit Code: 137", "Memory 97%"]
  • No source tracking
  • No reliability signal
  • Just LLM's own judgment

Proposed (Validation Pipeline):
  • Structured objects with fact, why, source, reliability
  • Dual validation: PATH A (assertions) + PATH B (rules)
  • User-friendly + traceable

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

🏗️ TWO-MODEL ARCHITECTURE

EvidenceItem (Internal - 12 fields)
  └─> Store in DB (complete metadata)
       └─> Select top 3-5
            └─> Transform to Evidence (UI - 4 fields)
                 └─> Return in API

Principle: Store complete, Show minimal

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📦 MODELS

1️⃣ EvidenceItem (Internal - Complete)
   • id, source, type, strength, relationship, availability
   • rawSnippet, reasoning, confidence, priority
   • collectedAt, metadata
   • Location: com.causa.core.domain.validation.EvidenceItem

2️⃣ Evidence (UI - Simplified)
   • evidenceFact: "Container exit code 137"
   • explanation: "Killed by Linux OOM killer"
   • source: "Kubernetes" (user-friendly)
   • rawSnippet: "Exit Code: 137"
   • reliability: "DEFINITIVE"
   • Location: com.causa.api.dto.response.DiagnosticDetailResponse.Evidence

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

🔄 API CHANGE

Current:
  List<String> evidences

New:
  List<Evidence> evidences

Example Response:
{
  "diagnosis": {
    "issue_title": "OOM Kill Detected",
    "evidences": [
      {
        "evidenceFact": "Container exit code 137",
        "explanation": "Killed by Linux OOM killer",
        "source": "Kubernetes",
        "rawSnippet": "Exit Code: 137",
        "reliability": "DEFINITIVE"
      },
      {
        "evidenceFact": "Heap usage 97% (500MB/512MB)",
        "explanation": "Severe memory pressure before crash",
        "source": "Prometheus",
        "rawSnippet": "jvm_memory_used_bytes{area=\"heap\"} 524288000",
        "reliability": "STRONG"
      }
    ]
  }
}

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

💾 STORAGE

DiagnosticEntity:
  • allEvidence: JSON (List<EvidenceItem>) ← Full 12-field model

At runtime:
  1. Load all evidence from DB
  2. Select top 3-5 by priority
  3. Transform: EvidenceItem → Evidence
  4. Return in API

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

✅ BENEFITS

• User-Focused: 4 simple fields vs 12 technical
• Clear: "fact" + "why" structure
• Traceable: Full model in DB for debugging
• Verified: From validation pipeline, not RCA LLM
• Transparent: Can show missing evidence
• Flexible: Can extend internal model without UI changes
• Source Tracking: User-friendly source names
